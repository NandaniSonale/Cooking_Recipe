project:
  name: "Cooking Recipe Management System"

description: >
  The Cooking Recipe Management System is a web-based application developed
  to manage and organize cooking recipes efficiently. The system allows users
  to add, view, and manage recipes in a structured manner. This project
  demonstrates CRUD operations and practical implementation of database
  management concepts using modern frontend development tools.

objectives:
  - Provide a structured platform to manage cooking recipes
  - Demonstrate CRUD operations (Create, Read, Update, Delete)
  - Apply database management concepts
  - Build an interactive and user-friendly interface

technologies_used:
  frontend:
    - HTML
    - CSS
    - JavaScript
  build_tool:
    - Vite
  runtime_environment:
    - Node.js
  package_manager:
    - npm

project_structure:
  - src/ (Source files)
  - node_modules/ (Installed dependencies)
  - index.html (Main HTML file)
  - package.json (Project configuration file)
  - package-lock.json
  - vite.config.ts (Vite configuration file)

features:
  - Add new recipes
  - View recipe details
  - Manage recipe data
  - Organized data structure
  - Fast development server using Vite

execution_steps:
  prerequisites:
    - Install Node.js (LTS version) from https://nodejs.org
    - Verify installation using:
        - node -v
        - npm -v

  steps:
    - Step 1: Clone the repository
        command: git clone https://github.com/Nandanisonale/Cooking-Recipe-Management.git

    - Step 2: Navigate to the project directory
        command: cd Cooking-Recipe-Management

    - Step 3: Install dependencies
        command: npm install

    - Step 4: Run development server
        command: npm run dev

    - Step 5: Open browser and visit
        url: http://localhost:3000/

    - To stop server:
        command: Ctrl + C
